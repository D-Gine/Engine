121void Server::add_handler(HttpMethod method, const std::string &pattern, Handler handler) {
122 Handler wrapped = wrap_handler(handler);
123
124 switch (method) {
125 case HttpMethod::GET:
126 svr_.Get(pattern.c_str(), [wrapped](const httplib::Request &req, httplib::Response &res) {
127 wrapped(req, res);
128 });
129 break;
130 case HttpMethod::POST:
131 svr_.Post(pattern.c_str(), [wrapped](const httplib::Request &req, httplib::Response &res) {
132 wrapped(req, res);
133 });
134 break;
135 case HttpMethod::PUT:
136 svr_.Put(pattern.c_str(), [wrapped](const httplib::Request &req, httplib::Response &res) {
137 wrapped(req, res);
138 });
139 break;
140 case HttpMethod::DELETE:
141 svr_.Delete(pattern.c_str(), [wrapped](const httplib::Request &req, httplib::Response &res) {
142 wrapped(req, res);
143 });
144 break;
145 case HttpMethod::PATCH:
146 svr_.Patch(pattern.c_str(), [wrapped](const httplib::Request &req, httplib::Response &res) {
147 wrapped(req, res);
148 });
149 break;
150 case HttpMethod::OPTIONS:
151 svr_.Options(pattern.c_str(), [wrapped](const httplib::Request &req, httplib::Response &res) {
152 wrapped(req, res);
153 });
154 break;
155 }

186void Server::start() {
187 if (running_)
188 return;
189 running_ = true;
190 std::println("Starting server on {}:{}", host_, port_);
191 if (!svr_.listen(host_.c_str(), port_)) {
192 running_ = false;
193 throw std::runtime_error("Failed to start server");
194 }
195}

197void Server::start_async() {
